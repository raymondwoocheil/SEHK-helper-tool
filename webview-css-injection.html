function isInAppWebView() {
    const userAgent = navigator.userAgent || navigator.vendor || window.opera;
    
    // Common patterns for mobile app webviews
    const webviewIndicators = [
        'wv',              // Android WebView
        'MobileApp',       // Generic mobile app indicator
        'WebView',         // Generic webview indicator
        'iPhone; CPU iPhone OS', // iOS WKWebView
    ];

    // Check if it's a mobile device
    const isMobile = /Android|iPhone|iPad|iPod/i.test(userAgent);
    
    if (!isMobile) return false;

    // Check for webview indicators
    const hasWebviewIndicator = webviewIndicators.some(indicator => 
        userAgent.includes(indicator)
    );

    // iOS-specific check
    const isIOS = /iPhone|iPad|iPod/i.test(userAgent);
    const isSafari = /Safari/i.test(userAgent);
    if (isIOS && !isSafari && userAgent.includes('CPU')) {
        return true;
    }

    return hasWebviewIndicator;
}

function injectCSS(isWebView) {
    let css;
    
    if (isWebView) {
        // CSS for in-app webview
        css = `
            .app-specific {
              display: block !important;
            }
            /* Add more webview-specific styles here */
        `;
    } else {
        // CSS for regular browser
        css = `
            .browser-specific {
              display: block !important;
            }
            /* Add more browser-specific styles here */
        `;
    }

    const styleElement = document.createElement('style');
    styleElement.type = 'text/css';
    styleElement.appendChild(document.createTextNode(css));
    document.head.appendChild(styleElement);
}

// Usage with debugging
const inWebView = isInAppWebView();
console.log('Webview status:', inWebView ? 'In App WebView' : 'Regular Browser');
injectCSS(inWebView);
